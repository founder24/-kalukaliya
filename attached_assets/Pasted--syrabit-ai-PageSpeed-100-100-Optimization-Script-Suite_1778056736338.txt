```javascript
// vite.config.js - Optimized configuration
import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'
import { VitePWA } from 'vite-plugin-pwa'
import compression from 'vite-plugin-compression'
import { resolve } from 'path'

export default defineConfig({
  plugins: [
    react(),
    // Gzip/Brotli compression for static assets
    compression({
      algorithm: 'brotliCompress',
      ext: '.br',
      threshold: 10240,
    }),
    // PWA for caching strategy (improves repeat visits)
    VitePWA({
      registerType: 'autoUpdate',
      workbox: {
        globPatterns: ['**/*.{js,css,html,ico,png,svg,webp,avif}'],
        runtimeCaching: [{
          urlPattern: /^https:\/\/.*\.(png|jpg|jpeg|svg|webp|avif)$/i,
          handler: 'CacheFirst',
          options: {
            cacheName: 'images-cache',
            expiration: { maxEntries: 50, maxAgeSeconds: 30 * 24 * 60 * 60 },
          },
        }],
      },
    }),
  ],
  build: {
    // Enable code-splitting via dynamic imports
    rollupOptions: {
      output: {
        manualChunks: {
          'vendor': ['react', 'react-dom', 'react-router-dom'],
          'ui-lib': ['@headlessui/react', '@heroicons/react'], // adjust to your libs
        },
        // Hash filenames for cache-busting
        entryFileNames: 'assets/[name].[hash].js',
        chunkFileNames: 'assets/[name].[hash].js',
        assetFileNames: 'assets/[name].[hash][extname]',
      },
    },
    // Minify with Terser for better JS compression
    minify: 'terser',
    terserOptions: {
      compress: {
        drop_console: true, // Remove console logs in production
        drop_debugger: true,
      },
    },
    // Generate sourcemaps only if needed for debugging
    sourcemap: false,
    // Target modern browsers to reduce polyfill bloat
    target: 'esnext',
  },
  // Preload critical assets
  optimizeDeps: {
    esbuildOptions: {
      target: 'esnext',
    },
  },
  // Server config for development (doesn't affect prod build)
  server: {
    port: 5173,
  },
})
```

## 🖼️ Step 3: Image Optimization Script (Node.js)

```javascript
// scripts/optimize-images.js
import sharp from 'sharp';
import { readdir, mkdir, stat } from 'fs/promises';
import { join, extname } from 'path';

const INPUT_DIR = 'public/images'; // Adjust to your image directory
const OUTPUT_DIR = 'public/images-optimized';
const QUALITY = 80; // Balance between size and quality

async function optimizeImage(inputPath, outputPath, format = 'webp') {
  try {
    const transformer = sharp(inputPath)
      .rotate() // Auto-rotate based on EXIF
      .resize({ width: 1920, withoutEnlargement: true }) // Max width cap
      .toFormat(format, { quality: QUALITY });

    await transformer.toFile(outputPath);
    console.log(`✅ Optimized: ${inputPath} → ${outputPath}`);
  } catch (err) {
    console.error(`❌ Failed ${inputPath}:`, err.message);
  }
}

async function processDirectory(dir) {
  const entries = await readdir(dir, { withFileTypes: true });
  
  for (const entry of entries) {
    const inputPath = join(dir, entry.name);
    
    if (entry.isDirectory()) {
      await mkdir(join(OUTPUT_DIR, entry.name), { recursive: true });
      await processDirectory(inputPath);
    } else if (entry.isFile() && /\.(png|jpe?g|webp)$/i.test(entry.name)) {
      const relativePath = dir.replace(INPUT_DIR, '');
      const outputSubdir = join(OUTPUT_DIR, relativePath);
      await mkdir(outputSubdir, { recursive: true });
      
      const baseName = entry.name.replace(/\.[^.]+$/, '');
      const outputPath = join(outputSubdir, `${baseName}.webp`);
      
      await optimizeImage(inputPath, outputPath);
    }
  }
}

// Main execution
(async () => {
  console.log('🚀 Starting image optimization...');
  await mkdir(OUTPUT_DIR, { recursive: true });
  await processDirectory(INPUT_DIR);
  console.log('✨ Optimization complete!');
})();
```

## 🧵 Step 4: Critical CSS Extraction (Post-Build)

```javascript
// scripts/extract-critical-css.js
import critical from 'critical';
import { readFileSync, writeFileSync } from 'fs';
import { resolve } from 'path';

const DIST_DIR = 'dist';
const INDEX_HTML = resolve(DIST_DIR, 'index.html');

async function extractCriticalCSS() {
  try {
    console.log('🔍 Extracting critical CSS...');
    
    const { css } = await critical.generate({
      inline: false,
      base: DIST_DIR,
      src: 'index.html',
      width: 375,   // Mobile viewport
      height: 667,
      penthouse: {
        forceInclude: ['.header', '.hero', '.nav'], // Adjust to your critical selectors
      },
    });

    // Read index.html and inject critical CSS
    let html = readFileSync(INDEX_HTML, 'utf8');
    
    // Insert critical CSS in <head> before other stylesheets
    const criticalStyle = `<style id="critical-css">${css}</style>`;
    html = html.replace('</head>', `${criticalStyle}\n</head>`);
    
    // Mark non-critical CSS for async loading
    html = html.replace(
      /<link rel="stylesheet" href="(.*\.css)"(?![^>]*id="critical")>/g,
      '<link rel="preload" href="$1" as="style" onload="this.rel=\'stylesheet\'"><noscript><link rel="stylesheet" href="$1"></noscript>'
    );
    
    writeFileSync(INDEX_HTML, html);
    console.log('✅ Critical CSS injected and non-critical CSS deferred');
    
  } catch (err) {
    console.error('❌ Critical CSS extraction failed:', err.message);
  }
}

extractCriticalCSS();
```

## 🔑 Key Trade-Offs & Notes for syrabit.ai

| Optimization | Benefit | Trade-Off | Mitigation |
|-------------|---------|-----------|------------|
| **Critical CSS inlining** | Faster FCP/LCP on mobile | Slightly larger initial HTML | Only inline above-the-fold CSS; async-load the rest |
| **Aggressive image compression** | Smaller payloads, faster LCP | Potential quality loss at very low sizes | Use `quality: 80` as baseline; test visually |
| **Code-splitting vendor chunks** | Faster parsing, better caching | More HTTP requests (mitigated by HTTP/2) | Use `manualChunks` wisely; preload critical chunks |
| **Removing console logs** | Smaller JS bundle | Harder debugging in prod | Keep sourcemaps in staging; use `drop_console: true` only in prod |
| **SSR/SSG migration** *(future step)* | Best possible LCP/FCP | Increased build complexity, server requirements | Start with SSG for static pages; keep SPA for dynamic sections |

## 📊 Post-Optimization Validation

After deploying the optimized build:

1. **Run PageSpeed Insights again** for both mobile and desktop
2. **Check Core Web Vitals in Chrome DevTools** → Lighthouse tab
3. **Monitor real-user metrics** via your analytics (if using RUM)
4. **Test on actual mobile devices** – emulators don't capture everything

If you're still not at 100/100, the remaining gaps are likely:
- Third-party script blocking (defer analytics, chat widgets)
- Font loading strategy (use `font-display: swap`, preload critical fonts)
- Server response time (ensure FastAPI backend is cached/optimized)
